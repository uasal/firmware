36--Considering this as a REGISTER_IN_BITS-bit output A/D for rough compatibility with AD7760, we have 8 bits left for PPS timestamp.
37--100MHz / 256 (8b) < 400k == 2^19 * 0.75, so we have 13b left for seconds of day, which would repeat every 2.5 hours.
38--And we're writing the actual seconds into the datafile headers, so we'll have to tweak it up in the C code. 100MHz = 27 bits.

92 constant MaxVal : std_logic_vector((REGISTER_IN_BITS - 1) downto (REGISTER_IN_BITS - DATA_BITS)) := (others => '1');
93
94 signal DitherClock : std_logic;
95 signal LastDitherClock : std_logic;
96
97 signal DitherPos : natural range 0 to ((2**DITHER_BITS) - 1) := 0;
98
99 signal DacMagnitudeOutA_i : std_logic_vector((REGISTER_IN_BITS - 1) downto 0);
100 signal DacMagnitudeOutB_i : std_logic_vector((REGISTER_IN_BITS - 1) downto 0);

104begin
105
106 --~ TP1 <= std_logic_vector(to_unsigned(DitherPos, DITHER_BITS))(0);
107 --~ TP2 <= std_logic_vector(to_unsigned(DitherPos, DITHER_BITS))(1);
108 TP1 <= DitherClock;
109 TP2 <= DitherClock;
110 TP3 <= DitherClock;
111 TP4 <= LastDitherClock;

113 --~ TP1 <= '0';
114 --~ TP2 <= '0';
115 --~ TP3 <= '0';
116 --~ TP4 <= '0';
117
118 DitherClk : ClockDividerPorts
119 generic map
120 (
121 CLOCK_DIVIDER => DITHER_CLOCK_DIV,
122 DIVOUT_RST_STATE => '0'--;
123 )
124 port map
125 (
126 clk => clk,
127 rst => rst,
128 div => DitherClock
129 );
130
131 DacTrigger <= not(DitherClock); --give us time to move things on rising edge so we don't have race conditions
132
133 DacMagnitudeOutA <= DacMagnitudeOutA_i;
134 DacMagnitudeOutB <= DacMagnitudeOutB_i;
135 DacMagnitudeOutC <= DacMagnitudeOutC_i;
136 DacMagnitudeOutD <= DacMagnitudeOutD_i;
137
138 --Read A/D:
139 process (clk, rst, DacXferComplete)
140 begin
141
142 if (rst = '1') then --We're using AdcClkReset instead of the external rst signal here so that sync will reset SamplesAveraged so our sample is aligned to sync when we are downsampling...
143
144 DitherPos <= 0;
145 LastDitherClock <= '0';
146 DacMagnitudeOutA_i <= x"BADCDE";
147 DacMagnitudeOutB_i <= x"BADCDE";
148 DacMagnitudeOutC_i <= x"BADCDE";
149 DacMagnitudeOutD_i <= x"BADCDE";
150
151 else
152
153 if ( (clk'event) and (clk = '1') ) then
154
155 if (DitherClock /= LastDitherClock) then
156
157 LastDitherClock <= DitherClock;
158
159 if (DitherClock = '1') then
160
161 if (DitherPos < ((2**DITHER_BITS) - 1)) then DitherPos <= DitherPos + 1; else DitherPos <= 0; end if;
162
163 if ( ((SetPointInA((REGISTER_IN_BITS - 1) - DATA_BITS downto (REGISTER_IN_BITS - DATA_BITS) - DITHER_BITS)) < std_logic_vector(to_unsigned(DitherPos, DITHER_BITS))) and (SetPointInA((REGISTER_IN_BITS - 1) downto (REGISTER_IN_BITS - DATA_BITS)) < MaxVal) ) then
164
165 DacMagnitudeOutA_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInA((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS) + std_logic_vector(to_unsigned(1, DATA_BITS));
166 DacMagnitudeOutA_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
167
168 else
169
170 DacMagnitudeOutA_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInA((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS);
171 DacMagnitudeOutA_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
172
173 end if;
174
175 if ( ((SetPointInB((REGISTER_IN_BITS - 1) - DATA_BITS downto (REGISTER_IN_BITS - DATA_BITS) - DITHER_BITS)) < std_logic_vector(to_unsigned(DitherPos, DITHER_BITS))) and (SetPointInB((REGISTER_IN_BITS - 1) downto (REGISTER_IN_BITS - DATA_BITS)) < MaxVal) ) then
176
177 DacMagnitudeOutB_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInB((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS) + std_logic_vector(to_unsigned(1, DATA_BITS));
178 DacMagnitudeOutB_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
179
180 else
181
182 DacMagnitudeOutB_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInB((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS);
183 DacMagnitudeOutB_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
184
185 end if;
186
187 if ( ((SetPointInC((REGISTER_IN_BITS - 1) - DATA_BITS downto (REGISTER_IN_BITS - DATA_BITS) - DITHER_BITS)) < std_logic_vector(to_unsigned(DitherPos, DITHER_BITS))) and (SetPointInC((REGISTER_IN_BITS - 1) downto (REGISTER_IN_BITS - DATA_BITS)) < MaxVal) ) then
188
189 DacMagnitudeOutC_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInC((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS) + std_logic_vector(to_unsigned(1, DATA_BITS));
190 DacMagnitudeOutC_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
191
192 else
193
194 DacMagnitudeOutC_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInC((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS);
195 DacMagnitudeOutC_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
196
197 end if;
198
199 if ( ((SetPointInD((REGISTER_IN_BITS - 1) - DATA_BITS downto (REGISTER_IN_BITS - DATA_BITS) - DITHER_BITS)) < std_logic_vector(to_unsigned(DitherPos, DITHER_BITS))) and (SetPointInD((REGISTER_IN_BITS - 1) downto (REGISTER_IN_BITS - DATA_BITS)) < MaxVal) ) then
200
201 DacMagnitudeOutD_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInD((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS) + std_logic_vector(to_unsigned(1, DATA_BITS));
202 DacMagnitudeOutD_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
203
204 else
205
206 DacMagnitudeOutD_i((REGISTER_OUT_BITS - 1) downto REGISTER_OUT_BITS - DATA_BITS) <= SetPointInD((REGISTER_IN_BITS - 1) downto REGISTER_IN_BITS - DATA_BITS);
207 DacMagnitudeOutD_i((REGISTER_OUT_BITS - 1) - DATA_BITS downto 0) <= (others => '0');
208
209 end if;
210
211 end if;
212
213 end if;
214
215 end if;
216
217 end if;
218
219 end process;

221end DacDithererLogic;
