178 process (clk, rst)
179 begin
180
181 Dbg2 <= LastWriteReq;
182 Dbg3 <= WriteReq;
183
184 DataStartAddress <= DataStartAddress_i;
185 DataEndAddress <= WriteAddress;
186
187 Empty <= '1' when (DataStartAddress_i = WriteAddress) else '0';
188
189 Full <= '1' when ( ( (DataStartAddress_i = 0) and (WriteAddress = ((2**PeekRamDepth) - 1)) ) or
190 ( (DataStartAddress_i = 1) and (WriteAddress = 0) ) or
191 ( (DataStartAddress_i > 1) and (WriteAddress = (DataStartAddress_i - 1)) )
192 )
193 else '0';
194
195 Count <= (WriteAddress - DataStartAddress_i) when (WriteAddress >= DataStartAddress_i) else
196 ( ((2**PeekRamDepth) - WriteAddress) - DataStartAddress_i);
197
198 if (rst = '1') then
199
200 LastPopReq <= '0';
201 LastWriteReq <= '0';
202 DataStartAddress_i <= (others => '0');
203 WriteAddress <= (others => '0');
204 LastHeaderEnd <= (others => '0');
205 LastFooterEnd <= (others => '0');
206 PayloadLen <= x"00000000";
207
208 else

210
211 LastPopReq <= PopReq;
212 LastWriteReq <= WriteReq;
213 LastHeaderFound <= HeaderFound;
214 LastFooterFound <= FooterFound;
215
216 if ( (LastPopReq = '0') and (PopReq = '1') ) then
217
218 DataStartAddress_i <= PopAddress;
219
220 end if;
221
222 if ( (LastWriteReq = '0') and (WriteReq = '1') ) then
223
224 Dbg1 <= '1';
225
226 if (WriteAddress < ( (2**PeekRamDepth) - 1) ) then
227
228 WriteAddress <= WriteAddress + std_logic_vector(to_unsigned(1, PeekRamDepth));
229 --~ WriteAddress <= WriteAddress + "0000000001";
230
231 else --wrap
232
233 WriteAddress <= (others => '0');
234
235 end if;
236
237 --If we wrap footer or header, clear!
238 if (LastHeaderEnd = WriteAddress + "0000000001") then LastHeaderEnd <= (others => '0'); end if;
239
240 if (LastFooterEnd = WriteAddress + "0000000001") then LastFooterEnd <= (others => '0'); end if;
241
242 --Grab the payload len?
243
244 if (WriteAddress >= LastHeaderEnd) then
245
246 if (WriteAddress = (LastHeaderEnd + 3) then PayloadLen <= MaybePayloadLen; end if;
247
248 else
249
250 if (WriteAddress = (LastHeaderEnd + 3 - (2**PeekRamDepth)) then PayloadLen <= MaybePayloadLen; end if;
251
252 end if;
253
254 else
255
256 Dbg1 <= '0';
257
258 end if;
259
260 --Update on the edge of found; can't put this on the writereq edge, because the flag will toggle on the next clock after, not synchrounously!
261 if ( (LastHeaderFound = '0') and (HeaderFound = '1') ) then LastHeaderEnd = WriteAddress - "0000000001"; end if;
262
263 if ( (LastFooterFound = '0') and (FooterFound = '1') ) then
264
265 LastFooterEnd = WriteAddress - "0000000001";
266

268 --Really need to test & fix CRCer first...

283 --~ CrcComplete => Uart0CrcDone--,
284 --~ );

286 --~ HeaderFooterPayloadLenMatches <=
287
288 end if;

291 end if;
