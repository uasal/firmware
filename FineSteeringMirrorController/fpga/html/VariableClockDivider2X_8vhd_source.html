45architecture VariableClockDivider2X of VariableClockDivider2XPorts is
46 signal ClkDiv : natural range 0 to ((2**WIDTH_BITS) - 1);
47 signal clko_i : std_logic;
48begin
49
50 clko <= clko_i;
51
52 process (clki, rst, rst_count)
53 begin
54
55 if (rst = '1') then
56
57 ClkDiv <= to_integer(unsigned(rst_count));
58 clko_i <= DIVOUT_RST_STATE;
59
60 else
61
62 if ( (clki'event) and (clki = '1') ) then
63
64 --Generate the master DE-sampling clock
65 if (ClkDiv < terminal_count) then
66
67 ClkDiv <= ClkDiv + 1;
68
69 else
70
71 ClkDiv <= 0;
72
73 end if;
74
75 if (ClkDiv < ( (shift_right(unsigned(terminal_count),1)))) or
76 (ClkDiv >= terminal_count) then
77
78 clko_i <= DIVOUT_RST_STATE;
79
80 else
81
82 clko_i <= not(DIVOUT_RST_STATE);
83
84 end if;
85
86 end if;
87
88 end if;
89
90end process;

92end VariableClockDivider2X;
