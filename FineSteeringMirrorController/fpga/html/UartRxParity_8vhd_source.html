43architecture Behaviour of UartRxParity is
44
45 signal RReg : std_logic_vector(7 downto 0); -- receive register
46 signal ParityBit : std_logic := '0';
47 signal ParityErrFailed : std_logic := '0';
48 signal RxDPrev : std_logic := '1';
49
50begin
51
52 -- Rx Process
53 RxProc : process(Clk, Reset)
54
55 variable BitPos : INTEGER range 0 to 11; -- Position of the bit in the frame
56 variable SampleCnt : INTEGER range 0 to 15; -- Count from 0 to 15 in each bit
57
58begin
59 if Reset = '1' then -- Reset
60 BitPos := 0;
61 SampleCnt := 0;
62 RxAv <= '0';
63 RReg <= x"00";
64 DataO <= x"00";
65 ParityErr <= '0';
66 ParityBit <= '0';
67 ParityErrFailed <= '0';
68 RxDPrev <= '1';
69 elsif Rising_Edge(Clk) then
70 if Enable = '1' then
71 case BitPos is
72 when 0 => -- idle
73 if (RxDPrev = '1') and (RxD = '0') then -- Start bit falling edge
74 RxAv <= '0';
75 SampleCnt := 0;
76 BitPos := 1;
77 ParityBit <= '0';
78 --~ Start <= '1';
79 end if;
80 when 1 => -- Start Bit
81 if SampleCnt = 7 then
82 if RxD = '0' then
83 null;
84 else
85 BitPos := 0; -- Glitch: abandon frame and return idle
86 end if;
87 elsif SampleCnt = 15 then
88 BitPos := 2;
89 end if;
90 when 10 => -- Parity
91 --~ Samp <= '0';
92 if SampleCnt = 7 then
93 if (PARITY_EVEN = 0 and RxD /= ParityBit) or
94 (PARITY_EVEN = 1 and RxD /= not ParityBit) then
95 ParityErrFailed <= '1';
96 else
97 ParityErrFailed <= '0';
98 end if;
99 end if;
100 if SampleCnt = 15 then
101 BitPos := BitPos + 1;
102 end if;
103 when 11 => -- Stop Bit
104 if (SampleCnt >= 3) then
105 if (Rxd = '1') and (ParityErrFailed = '0') then --stop bit
106 BitPos := 0; -- next is idle
107 RxAv <= '1';
108 DataO <= RReg; -- Store received byte
109 ParityErr <= '0';
110 elsif (Rxd = '1') and (ParityErrFailed = '1') then --stop bit but parity error
111 BitPos := 0; -- next is idle
112 RxAv <= '1';
113 DataO <= RReg; -- Store received byte
114 ParityErr <= '1';
115 end if;
116 end if;
117 if (SampleCnt >= 13) then --no stop bit
118 BitPos := 0; -- next is idle
119 end if;
120 when others =>
121 RxAv <= '0';
122 ParityErrFailed <= '0';
123 if SampleCnt = 7 and BitPos >= 2 and BitPos <= 9 then -- data bits
124 RReg(BitPos-2) <= RxD; -- Deserialisation
125 ParityBit <= ParityBit xor RxD;
126 end if;
127 if SampleCnt = 15 then -- Increment BitPos on 3
128 BitPos := BitPos + 1;
129 end if;
130 end case;
131 if SampleCnt = 15 then
132 SampleCnt := 0;
133 else
134 sampleCnt := SampleCnt + 1;
135 end if;
136 RxDPrev <= RxD;
137 end if;
138 end if;
139 end process;
140end Behaviour;
