134void CGraphFSMMonitorAdc::Process()
135{
136 if (AdcFound)
137 {
138 //~ we don't have anything connected to gpios but might in the future, and dm does if this code gets cut&pasted: GpioInputs = Adc.ReadRegister(ads1258details::register_gpiod);
139
140 ads1258details::ads1258sample sample;
141
142 for(size_t CurrentChan = 0; CurrentChan < ads1258details::ads1258numchannels; CurrentChan++)
143 {
144 ProcessAllUarts();
145
146 Adc.AutoScan();
147
148 if (Adc.IsScanChannel(CurrentChan))
149 {
150 ProcessAllUarts();
151
152 Adc.GetLastSample(CurrentChan, sample);
153
154 ProcessAllUarts();
155
156 if ( (sample.status.isbrownout) || (sample.status.isclipped) )
157 {
158 if (Monitor) { ::formatf("\nMonitorAdc: ch %u bad status: 0x%.2X\n", sample.status.channel, sample.status.all); }
159 }
160 else
161 {
162 if (sample.status.isnew)
163 {
164 if (Monitor) { ::formatf("\nMonitorAdc: ch %u reads: %lf Volts [%lu lsb's]\n", sample.status.channel, Adc.CountsToVolts(sample.sample), sample.sample); }
165
166 switch(sample.status.channel)
167 {
168 case IHVChannel : { P1V2 = (sample.sample); break; }
169 case IMVChannel : { P2V2 = (sample.sample); break; }
170 case I6VChannel : { P28V = (sample.sample); break; }
171 case I3VDChannel : { P2V5 = (sample.sample); break; }
172 case I2VDChannel : { P6V = (sample.sample); break; }
173 case I1VChannel : { P5V = (sample.sample); break; }
174 case StrainBPChannel : { P3V3D = (sample.sample); break; }
175 case StrainBMChannel : { P4V3 = (sample.sample); break; }
176 case StrainDPChannel : { P2I2 = (sample.sample); break; }
177 case StrainDMChannel : { P4I3 = (sample.sample); break; }
178 case StrainCMChannel : { P6I = (sample.sample); break; }
179 case StrainCPChannel : { Aux0 = (sample.sample); break; }
180 case StrainAMChannel : { Aux1 = (sample.sample); break; }
181 case StrainAPChannel : { Aux2 = (sample.sample); break; }
182 case P5VDChannel : { AmbientLight = (sample.sample); break; }
183 case I2VAChannel : { Temperature = (sample.sample); break; }
184
185 default: { break; } //do nothing; we can scan all 16 channels and discard the unused ones.
186 }
187 }
188 else
189 {
190 if (Monitor) { ::formatf("\nMonitorAdc: ch %u no new sample this scan.\n", sample.status.channel); }
191 }
192 }
193 }
194 }
