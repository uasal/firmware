99void CGraphFSMMonitorAdc::Init()
100{
101 for (uint8_t i = 0; i < 10; i++)
102 {
103 uint8_t AdcInit = Adc.Init(GpioDirections);
104 if (Adc.InitOK == AdcInit)
105 {
106 AdcFound = true;
107 break;
108 }
109 else
110 {
111 formatf("\nMonitorAdc: Error initializing Adc: %u [0x%.2X].\n", AdcInit, AdcInit);
112 }
113 }
114 if (AdcFound)
115 {
116 Adc.ClearScanChannels();
117
118 //Adc0 channels:
119 Adc.AddScanChannel(IHVChannel);
120 Adc.AddScanChannel(INVChannel);
121 Adc.AddScanChannel(I6VChannel);
122 Adc.AddScanChannel(I3VDChannel);
123 Adc.AddScanChannel(I2VDChannel);
124 Adc.AddScanChannel(I1VChannel);
125 Adc.AddScanChannel(StrainBPChannel);
126 Adc.AddScanChannel(StrainBMChannel);
127 Adc.AddScanChannel(StrainBChannel);
128 Adc.AddScanChannel(StrainDPChannel);
129 Adc.AddScanChannel(StrainDMChannel);
130 Adc.AddScanChannel(StrainDChannel);
131 Adc.AddScanChannel(StrainCMChannel);
132 Adc.AddScanChannel(StrainCPChannel);
133 Adc.AddScanChannel(StrainCChannel);
134 Adc.AddScanChannel(StrainAMChannel);
135 Adc.AddScanChannel(StrainAPChannel);
136 Adc.AddScanChannel(StrainAChannel);
137 Adc.AddScanChannel(P5VDChannel);
138 Adc.AddScanChannel(I2VAChannel);
139
140 //Adc1 channels:
141 Adc.AddScanChannel(TempChannel);
142 Adc.AddScanChannel(P3V3DChannel);
143 Adc.AddScanChannel(P28VChannel);
144 Adc.AddScanChannel(P2V2Channel);
145 Adc.AddScanChannel(P2V5DChannel);
146 Adc.AddScanChannel(P1V2Channel);
147 Adc.AddScanChannel(P2V5AChannel);
148 Adc.AddScanChannel(P4V3Channel);
149 Adc.AddScanChannel(I3VAChannel);
150 Adc.AddScanChannel(P3V3AChannel);
151 Adc.AddScanChannel(P6VChannel);
152 Adc.AddScanChannel(P5VAChannel);
153 Adc.AddScanChannel(LuxRadsChannel);
154 Adc.AddScanChannel(N18VChannel);
155 Adc.AddScanChannel(N20VChannel);
156 Adc.AddScanChannel(P125VChannel);
157
158 Adc.CommitScanChannels();
159 Adc.StartChannelScan();
160 //Take start pin high to initiate auto-scan
161 {
162 Adc.WriteRegister(ads1258details::register_gpiod, StartPinMask);
163 //~ if (Monitor)
164 {
165 formatf("\nMonitorAdc: Starting A/D auto-scan; wrote 0x%.2X to gpio's, readback: 0x%.2X\n", StartPinMask, Adc.ReadRegister(ads1258details::register_gpiod));
166 fflush(stdout);
167 }
168 }
169 }
170 else
171 {
172 formatf("\nMonitorAdc: No Adc found!\n");
173 }

176void CGraphFSMMonitorAdc::Process()
177{
178 if (AdcFound)
179 {
180 //~ we don't have anything connected to gpios but might in the future, and dm does if this code gets cut&pasted: GpioInputs = Adc.ReadRegister(ads1258details::register_gpiod);
181
182 ads1258details::ads1258sample sample0;
183 ads1258details::ads1258sample sample1;
184
185 for(size_t CurrentChan = 0; CurrentChan < ads1258details::ads1258numchannels; CurrentChan++)
186 {
187 ProcessAllUarts();
188
189 Adc.AutoScan();
190
191 if (Adc.IsScanChannel(CurrentChan))
192 {
193 ProcessAllUarts();
194
195 Adc.GetLastSample(CurrentChan, sample0, sample1);
196
197 ProcessAllUarts();
198
199 if ( (sample0.status.isbrownout) || (sample0.status.isclipped) || (sample1.status.isbrownout) || (sample1.status.isclipped) )
200 {
201 if (Monitor) { ::formatf("\nMonitorAdc: bad status: Adc0: ch %u: 0x%.2X, Adc1: ch %u: 0x%.2X\n", sample0.status.channel, sample0.status.all, sample1.status.channel, sample1.status.all); }
202 }
203 else
204 {
205 if ( (sample0.status.isnew) || (sample1.status.isnew) )
206 {
207 if (Monitor) { ::formatf("\nMonitorAdc: Adc0: ch %u reads: %lf Volts [%lu lsb's]; Adc1: ch %u reads: %lf Volts [%lu lsb's]\n", sample0.status.channel, Adc.CountsToVolts(sample0.sample), sample0.sample, sample1.status.channel, Adc.CountsToVolts(sample1.sample), sample1.sample); }
208
209 switch(sample0.status.channel)
210 {
211 //Adc0 channels:
212 case IHVChannel : { IHV = (sample0.sample); break; }
213 case INVChannel : { INV = (sample0.sample); break; }
214 case I6VChannel : { I6V = (sample0.sample); break; }
215 case I3VDChannel : { I3VD = (sample0.sample); break; }
216 case I2VDChannel : { I2VD = (sample0.sample); break; }
217 case I1VChannel : { I1V = (sample0.sample); break; }
218 case StrainBPChannel : { StrainBP = (sample0.sample); break; }
219 case StrainBMChannel : { StrainBM = (sample0.sample); break; }
220 case StrainBChannel : { StrainB = (sample0.sample); break; }
221 case StrainDPChannel : { StrainDP = (sample0.sample); break; }
222 case StrainDMChannel : { StrainDM = (sample0.sample); break; }
223 case StrainDChannel : { StrainD = (sample0.sample); break; }
224 case StrainCMChannel : { StrainCM = (sample0.sample); break; }
225 case StrainCPChannel : { StrainCP = (sample0.sample); break; }
226 case StrainCChannel : { StrainC = (sample0.sample); break; }
227 case StrainAMChannel : { StrainAM = (sample0.sample); break; }
228 case StrainAPChannel : { StrainAP = (sample0.sample); break; }
229 case StrainAChannel : { StrainA = (sample0.sample); break; }
230 case P5VDChannel : { P5VD = (sample0.sample); break; }
231 case I2VAChannel : { I2VA = (sample0.sample); break; }
232 default: { break; } //do nothing; we can scan all 16 channels and discard the unused ones.
233 }
234
235 switch(sample1.status.channel)
236 {
237 //Adc1 channels:
238 case TempChannel : { Temp = (sample1.sample); break; }
239 case P3V3DChannel : { P3V3D = (sample1.sample); break; }
240 case P28VChannel : { P28V = (sample1.sample); break; }
241 case P2V2Channel : { P2V2 = (sample1.sample); break; }
242 case P2V5DChannel : { P2V5D = (sample1.sample); break; }
243 case P1V2Channel : { P1V2 = (sample1.sample); break; }
244 case P2V5AChannel : { P2V5A = (sample1.sample); break; }
245 case P4V3Channel : { P4V3 = (sample1.sample); break; }
246 case I3VAChannel : { I3VA = (sample1.sample); break; }
247 case P3V3AChannel : { P3V3A = (sample1.sample); break; }
248 case P6VChannel : { P6V = (sample1.sample); break; }
249 case P5VAChannel : { P5VA = (sample1.sample); break; }
250 case LuxRadsChannel : { LuxRads = (sample1.sample); break; }
251 case N18VChannel : { N18V = (sample1.sample); break; }
252 case N20VChannel : { N20V = (sample1.sample); break; }
253 case P125VChannel : { P125V = (sample1.sample); break; }
254 default: { break; } //do nothing; we can scan all 16 channels and discard the unused ones.
255 }
256 }
257 else
258 {
259 if (Monitor) { ::formatf("\nMonitorAdc: ch %u no new sample this scan.\n", sample0.status.channel); }
260 }
261 }
262 }
263 }
264 }
