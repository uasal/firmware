86{

101int8_t BinaryFSMAdcsFloatingPointCommand(const uint32_t Name, char const* Params, const size_t ParamsLen, const void* Argument)
102{
103 printf("\nBinaryFSMAdcsCommand processing(%u)...\n\n", ParamsLen);
104
105 double AdcVals[3];
106 //Prepare for atomic read of samples (do as an 8-bit pointer so the processor doesn't crash !#@$%#!):
107 *((uint8_t*)&(FSM->LatchAdcs)) = 1;
108 AdcAccumulator A, B, C;
109 //~ A = FSM->AdcAAccumulator;
110 //~ B = FSM->AdcBAccumulator;
111 //~ C = FSM->AdcCAccumulator;
112 A.Samples = FSM->AdcAAccumulator;
113 A.SetHiWord(FSM->AdcAAccumulatorHiandNumAccums);
114 B.Samples = FSM->AdcBAccumulator;
115 B.SetHiWord(FSM->AdcBAccumulatorHiandNumAccums);
116 C.Samples = FSM->AdcCAccumulator;
117 C.SetHiWord(FSM->AdcCAccumulatorHiandNumAccums);
118 AdcVals[0] = (4.096 * (((double)A.Samples - 0) / (double)A.NumAccums)) / 8388608.0;
119 AdcVals[1] = (4.096 * (((double)B.Samples - 0) / (double)B.NumAccums)) / 8388608.0;
120 AdcVals[2] = (4.096 * (((double)C.Samples - 0) / (double)C.NumAccums)) / 8388608.0;
121 printf("\nBinaryFSMAdcsFloatingPointCommand Replying (%lf, %lf, %lf)...\n\n", AdcVals[0], AdcVals[1], AdcVals[2]);
122 TxBinaryPacket(Argument, CGraphPayloadTypeFSMAdcsFloatingPoint, 0, AdcVals, 3 * sizeof(double));
123
124 return(ParamsLen);
125}

127int8_t BinaryFSMDacsFloatingPointCommand(const uint32_t Name, char const* Params, const size_t ParamsLen, const void* Argument)
128{
129 double VA = 0.0, VB = 0.0, VC = 0.0;
130 unsigned long A = 0, B = 0, C = 0;
131
132 if (ParamsLen >= (3 * sizeof(double)))
133 {
134 const double* DacSetpoints = (const double*)Params;
135
136 VA = DacSetpoints[0];
137 VB = DacSetpoints[1];
138 VC = DacSetpoints[2];
139
140 A = (VA * (double)(0x00FFFFFFUL) * 60.0) / 4.096;
141 B = (VB * (double)(0x00FFFFFFUL) * 60.0) / 4.096;
142 C = (VC * (double)(0x00FFFFFFUL) * 60.0) / 4.096;
143
144 printf("\n\nBinaryFSMDacsCommand: Setting to: %3.1lf (%lx), %3.1lf (%lx), %3.1lf (%lx).\n", VA, A, VB, B, VC, C);
145
146 FSM->DacASetpoint = DacSetpoints[0];
147 FSM->DacBSetpoint = DacSetpoints[1];
148 FSM->DacCSetpoint = DacSetpoints[2];
149 }
150
151 uint32_t DacSetpoints[3];
152
153 A = FSM->DacASetpoint;
154 B = FSM->DacBSetpoint;
155 C = FSM->DacCSetpoint;
156
157 VA = 4.096 * (double)(A) / ((double)(0x00FFFFFFUL) * 60.0);
158 VB = 4.096 * (double)(B) / ((double)(0x00FFFFFFUL) * 60.0);
159 VC = 4.096 * (double)(C) / ((double)(0x00FFFFFFUL) * 60.0);
160
161 DacSetpoints[0] = VA;
162 DacSetpoints[1] = VB;
163 DacSetpoints[2] = VC;
164
165 printf("\n\nBinaryFSMDacsCommand: Replying: %3.1lf (%lx), %3.1lf (%lx), %3.1lf (%lx).\n", VA, A, VB, B, VC, C);
166 TxBinaryPacket(Argument, CGraphPayloadTypeFSMDacsFloatingPoint, 0, DacSetpoints, 3 * sizeof(double));
167
168 return(ParamsLen);
169}
