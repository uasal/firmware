43 static bool busy() { return(0 == (FSM->MonitorAdcSpiCommandStatusRegister.TransactionComplete) ); }
44 static void waitbusytimeout()
45 {
46 size_t i = 0;
47 for(i = 0; i < spi_timeout; i++)
48 {
49 if (!busy()) { break; }
50 delayus(1);
51 }
52 if (i >= spi_timeout - 2) { formatf("\nPinoutMonitorAdc: T/O."); }
53 }
54
55 static void enable(const bool en) { FSM->MonitorAdcSpiCommandStatusRegister.all = (uint32_t)en; }

58 {
59 waitbusytimeout();
60 FSM->MonitorAdcSpiTransactionRegister = (uint32_t)val;
61 }
62
63 static uint8_t receive(uint8_t val)
64 {
65 uint32_t out = 0;
66
67 //Do the xfer
68 waitbusytimeout();
69 FSM->MonitorAdcSpiTransactionRegister = (uint32_t)val;
70
71 //readback
72 waitbusytimeout();
73 out = FSM->MonitorAdcSpiTransactionRegister;
74
75 return(out & 0x000000FFUL);
76 }
