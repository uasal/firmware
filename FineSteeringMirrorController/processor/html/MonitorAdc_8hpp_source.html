43 static bool busy() { return(0 == (FSM->MonitorAdcSpiCommandStatusRegister.TransactionComplete) ); }
44 static void waitbusytimeout()
45 {
46 size_t i = 0;
47 for(i = 0; i < spi_timeout; i++)
48 {
49 if (!busy()) { break; }
50 delayus(1);
51 }
52 if (i >= spi_timeout - 2) { formatf("\nPinoutMonitorAdc: T/O."); }
53 }

59 {
60 waitbusytimeout();
61 FSM->MonitorAdcSpiTransactionRegister = val;
62 }

65 {
66 uint32_t out = 0;
67
68 //Do the xfer

70 FSM->MonitorAdcSpiTransactionRegister = val;
71
72 //readback
73 waitbusytimeout();
74 out = FSM->MonitorAdcSpiTransactionRegister;
75
76 //~ return(out & 0x0000FFFFUL);
77 return(out);
78 }
