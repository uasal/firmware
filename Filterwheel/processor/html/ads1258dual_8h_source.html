317 uint8_t ConvertOnceDiff(uint8_t chanpos, uint8_t channeg, ads1258details::ads1258sample& Adc0Sample, ads1258details::ads1258sample& Adc1Sample)
318 {
319 //Set fixedchannel mode:
320 ads1258details::config0register config0(1, 1, 1, 0, 0, 1);
321 config0.muxmod = 1;
322 WriteRegister(ads1258details::register_config0, config0.all);
323
324 //Set channel to read (always differential + & -)
325 ads1258details::muxschregister muxsch(chanpos, channeg);
326 WriteRegister(ads1258details::register_muxsch, muxsch.all);
327 uint16_t temp = 0;
328 ReadRegister(ads1258details::register_muxsch, temp);
329 if ( (muxsch.all != (temp & 0xFF)) || (muxsch.all != (temp >> 8)) )
330 {
331 #ifdef DEBUGADC
332 formatf("ads1258::ConvertOnceDiff(): muxsch mismatch (reads:0x%.4X)\n", temp);
333 #endif
334 return(ads1258details::register_muxdif);
335 }
336
337 //Convert:
338 SendCommand(ads1258details::ads1258cmdheader(ads1258details::cmdtype_pulseconvert, false, 0));
339
340 //Wait
341 delayus(1000000);
342
343 //Read data & return:
344 ReadLastChannel(Adc0Sample, Adc1Sample);
345
346 return(0);
347 }

474 bool Scan()
475 {
476 ads1258details::ads1258sample Adc0;
477 ads1258details::ads1258sample Adc1;
478 ReadLastChannel(Adc0, Adc1);
479
480 SendCommand(ads1258details::ads1258cmdheader(ads1258details::cmdtype_pulseconvert, false, 0));
481
482 if ( (Adc0.status.channel < ads1258details::ads1258numchannels) && (Adc0.status.isnew) )
483 {
484 lastsamples0[Adc0.status.channel] = Adc0;
485 }
486
487 if ( (Adc1.status.channel < ads1258details::ads1258numchannels) && (Adc1.status.isnew) )
488 {
489 lastsamples1[Adc1.status.channel] = Adc1;
490 }
491
492 bool Sucess = true;
493 return(Sucess);
494 }

530 virtual uint8_t Init(uint8_t gpio_output_mask = gpio_output_mask_all_outputs) //All outputs is reccommended in datasheet as inputs are not allowed to float for some ass reason.
531 {
532 //~ uint8_t Return = InitOK;
533 //see pg 41 of ads1258 revD datasheet.
534
535 //1. Reset spi - disable for 4096xFclk
536 {
537 enable(false);
538 delayus(10000);
539 }
540
541 //2. stop the converter: set "start" pin low
542 //<unimplemented in hardware, not strictly neccessary>
543
544 //3. Reset the converter
545 {
546 SendCommand(ads1258details::ads1258cmdheader(ads1258details::cmdtype_reset, false, 0));
547 delayus(1000);
548 }
549
550 //4. Config registers
551 //ads1258details::config0register config0(1, 0, 1, 0, 0, 1); //Chopper off
552 ads1258details::config0register config0(1, 0, 1, 0, 1, 1); //Chopper on
553 ads1258details::config1register config1(0, 7, 0, 0);
554 ads1258details::sysreadregister sysread(1, 1, 1, 1, 1);
555 {
556 WriteRegister(ads1258details::register_config0, config0.all);
557 WriteRegister(ads1258details::register_config1, config1.all);
558 WriteRegister(ads1258details::register_sysread, sysread.all);
559
560 //pg 42: set no-connected gpio's to outputs
561 WriteRegister(ads1258details::register_gpioc, gpio_output_mask); //all outputs
562 WriteRegister(ads1258details::register_gpiod, 0x00); //all zero
563 }
564
565 //5. Readback registers
566 {
567 uint16_t temp;
568
569 ReadRegister(ads1258details::register_config0, temp);
570 if ( (config0.all != (temp & 0xFF)) || (config0.all != (temp >> 8)) )
571 {
572 //~ #ifdef DEBUGADC
573 ::formatf("ads1258::init(): config0 mismatch (reads:0x%.4X, wanted:0x%.2X)\n", temp, config0.all);
574 //~ #endif
575 return(ads1258details::register_config0);
576 //~ Return |= ads1258details::register_config0;
577 }
578
579 ReadRegister(ads1258details::register_config1, temp);
580 if ( (config1.all != (temp & 0xFF)) || (config1.all != (temp >> 8)) )
581 {
582 //~ #ifdef DEBUGADC
583 ::formatf("ads1258::init(): config1 mismatch (reads:0x%.4X, wanted:0x%.2X)\n", temp, config1.all);
584 //~ #endif
585 return(ads1258details::register_config1);
586 //~ Return |= ads1258details::register_config1;
587 }
588
589 ReadRegister(ads1258details::register_sysread, temp);
590 if ( (sysread.all != (temp & 0xFF)) || (sysread.all != (temp >> 8)) )
591 {
592 //~ #ifdef DEBUGADC
593 ::formatf("ads1258::init(): sysread mismatch (reads:0x%.4X, wanted:0x%.2X)\n", temp, sysread.all);
594 //~ #endif
595 return(ads1258details::register_sysread);
596 //~ Return |= ads1258details::register_sysread;
597 }
598
599 //pg 42: set no-connected gpio's to outputs
600 ReadRegister(ads1258details::register_gpioc, temp);
601 if ( (gpio_output_mask != (temp & 0xFF)) || (gpio_output_mask != (temp >> 8)) )
602 {
603 //~ #ifdef DEBUGADC
604 ::formatf("ads1258::init(): gpioc mismatch (reads:0x%.4X, wanted:0x%.2X)\n", temp, gpio_output_mask);
605 //~ #endif
606 return(ads1258details::register_gpioc);
607 //~ Return |= ads1258details::register_gpioc;
608 }
609
610 //(Don't read gpiod - we write 0x00, but if pins are set as inputs, they will not read 0x00)
611
612 //Check and see if the ID register matched 0x8B:
613 ReadRegister(ads1258details::register_idnum, temp);
614 if ( (ads1258_idnum_always != (temp & 0xFF)) || (ads1258_idnum_always != (temp >> 8)) )
615 {
616 //~ #ifdef DEBUGADC
617 ::formatf("ads1258::init(): idnum mismatch (reads:0x%.4X, wanted:0x%.2X)\n", temp, 0x8B8B);
